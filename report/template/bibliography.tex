% IEEE numbering follows first citation, with one continuous reference list.
\newif\ifReportBiblatex
\ifdefined\UseBibtex
  \ReportBiblatexfalse
\else
  \IfFileExists{ieee.bbx}{\ReportBiblatextrue}{\ReportBiblatexfalse}
\fi
\providecommand{\ReportBooksDatabase}{bibliography/books}
\ifReportBiblatex
  \usepackage[backend=biber,style=ieee,sorting=none]{biblatex}
  \addbibresource{bibliography/legal.bib}
  \addbibresource{\ReportBooksDatabase.bib}
  \addbibresource{bibliography/articles.bib}
  \addbibresource{bibliography/references.bib}
  \DeclareLanguageMapping{vietnamese}{english}
  % ieee.cbx's cite command encloses each numeric citation in square brackets.
  \let\citep\cite
  \let\citet\textcite
  \newcommand{\PrintReportBibliography}{%
    \addchap{Tài liệu tham khảo}
    \begin{english}\printbibliography[heading=none]\end{english}}
\else
  % Standard TeX IEEE style for the BibTeX fallback; no grouped/reset numbering.
  \usepackage[numbers,square,sort&compress]{natbib}
  \renewcommand{\bibsection}{\addchap{Tài liệu tham khảo}}
  \newcommand{\PrintReportBibliography}{%
    \bibliographystyle{ieeetr}
    \bibliography{bibliography/legal,\ReportBooksDatabase,bibliography/articles,bibliography/references}}
\fi
