\addchap{Mở đầu}\label{chap:introduction}
\addsec{Tổng quan tình hình nghiên cứu trong nước và ngoài nước}
\DraftNote{Tổng hợp công trình có trích dẫn, so sánh phương pháp và chỉ ra khoảng trống nghiên cứu.}
\addsec{Lý do chọn đề tài}
\DraftNote{Liên hệ bài toán cứu hộ bão lũ, mạng yếu và nhu cầu điều phối.}
\addsec{Mục tiêu đề tài}
\DraftNote{Chép chính xác mục tiêu đăng ký, xác định tiêu chí đánh giá.}
\addsec{Phương pháp nghiên cứu}
\DraftNote{Mô tả phương pháp phát triển hệ thống và kiểm chứng thực nghiệm.}
\addsec{Đối tượng nghiên cứu}
\DraftNote{Xác định dữ liệu, mô hình và các thành phần hệ thống được nghiên cứu.}
\addsec{Phạm vi nghiên cứu}
\DraftNote{Giới hạn dữ liệu, nền tảng, điều kiện mạng và mức độ kiểm thử thực tế.}
